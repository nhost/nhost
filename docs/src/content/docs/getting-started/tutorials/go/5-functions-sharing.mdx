## Share a note with a collaborator

Add a `note_collaborators` table in the **SQL Editor**:

```sql SQL Editor
CREATE TABLE public.note_collaborators (
  note_id uuid NOT NULL REFERENCES public.notes (id) ON DELETE CASCADE,
  user_id uuid NOT NULL REFERENCES auth.users (id) ON DELETE CASCADE,
  role    text NOT NULL DEFAULT 'viewer' CHECK (role IN ('viewer', 'editor')),
  PRIMARY KEY (note_id, user_id)
);
```

Track it, then create **both** relationships **before** setting the filters
below (Database → each table → **Relationships**):

- a `collaborators` **array** relationship on `notes` → `note_collaborators`, and
- a `note` **object** relationship on `note_collaborators` → `notes`.

Order matters and the names matter: the `notes` filters reference
`collaborators` and the `note_collaborators` check references `note`, so if a
relationship is missing or mis-named the permission won't save
(`"collaborators" does not exist` / `"note" does not exist`).

Now update the `notes` permissions so collaborators can see and edit shared
notes — this is the one place the owner-only rule opens up:

- **notes → select** filter — owner **or** any collaborator:

  ```json
  { "_or": [
      { "user_id": { "_eq": "X-Hasura-User-Id" } },
      { "collaborators": { "user_id": { "_eq": "X-Hasura-User-Id" } } }
  ]}
  ```
- **notes → update** filter — owner **or** an `editor` collaborator:

  ```json
  { "_or": [
      { "user_id": { "_eq": "X-Hasura-User-Id" } },
      { "collaborators": { "_and": [
          { "user_id": { "_eq": "X-Hasura-User-Id" } },
          { "role": { "_eq": "editor" } }
      ]}}
  ]}
  ```

For **note_collaborators**, let a note's owner manage shares. This needs **four**
actions — `insert`, `select`, `update`, and `delete`:

- Check columns `note_id`, `user_id`, and `role` for `insert`, `select`, **and**
  `update`. The `update` permission on `role` is easy to miss but required: the
  `share` command below upserts with `on_conflict: { update_columns: [role] }`,
  and without it you hit the opaque
  `expected one of the values ['_PLACEHOLDER'] for type
  'note_collaborators_update_column'`.
- Use `{ "note": { "user_id": { "_eq": "X-Hasura-User-Id" } } }` as the `insert`
  check and the `select` / `update` / `delete` filter.

Add the commands (the owner shares by the collaborator's user id, defaulting to
viewer). `share` takes an optional positional `ROLE`:

```go title="main.go"
func cmdShare(ctx context.Context, c *nhost.Client, noteID, userID, role string) error {
	_, err := gql(ctx, c, `
		mutation Share($noteId: uuid!, $userId: uuid!, $role: String!) {
			insert_note_collaborators_one(
				object: {note_id: $noteId, user_id: $userId, role: $role}
				on_conflict: {constraint: note_collaborators_pkey, update_columns: [role]}
			) { note_id role }
		}`, graphql.Variables{"noteId": noteID, "userId": userID, "role": role})
	if err != nil {
		return err
	}
	fmt.Printf("shared %s with %s as %s\n", noteID, userID, role)
	return nil
}

func cmdUnshare(ctx context.Context, c *nhost.Client, noteID, userID string) error {
	_, err := gql(ctx, c, `
		mutation Unshare($noteId: uuid!, $userId: uuid!) {
			delete_note_collaborators_by_pk(note_id: $noteId, user_id: $userId) { note_id }
		}`, graphql.Variables{"noteId": noteID, "userId": userID})
	if err != nil {
		return err
	}
	fmt.Printf("unshared %s from %s\n", noteID, userID)
	return nil
}
```

Register both on the root command. `share` uses `RangeArgs(2, 3)` so `ROLE` is
optional and defaults to `viewer`; `unshare` deletes the collaborator row, which
the owner's `delete` permission on `note_collaborators` already allows:

```go title="main.go"
	root.AddCommand(&cobra.Command{
		Use:   "share NOTE_ID USER_ID [ROLE]",
		Short: "Share a note with another user (ROLE is viewer or editor)",
		Args:  cobra.RangeArgs(2, 3),
		RunE: func(_ *cobra.Command, args []string) error {
			role := "viewer"
			if len(args) > 2 {
				role = args[2]
			}
			return cmdShare(ctx, client, args[0], args[1], role)
		},
	})
	root.AddCommand(&cobra.Command{
		Use:   "unshare NOTE_ID USER_ID",
		Short: "Remove a collaborator from a note",
		Args:  cobra.ExactArgs(2),
		RunE: func(_ *cobra.Command, args []string) error {
			return cmdUnshare(ctx, client, args[0], args[1])
		},
	})
```

`export` is the one step that needs the function actually *running* — every part

`functions/notes/export.ts` is live at

## The export function

Serverless functions run on your backend. Create `functions/notes/export.ts` in
the `functions/` directory that `nhost init` scaffolded in Part 1 — file-based
routing maps it to `/v1/notes/export` (see the
[Functions docs](/products/functions)). It reads the caller's notes over
GraphQL, renders Markdown, and uploads the result to Storage — **forwarding the
caller's token**, so Hasura permissions and Storage rules apply exactly as they
do in the CLI:

```typescript title="functions/notes/export.ts"
import type { Request, Response } from "express";
import { createClient } from "@nhost/nhost-js";

const nhost = createClient({
  region: process.env.NHOST_REGION,
  subdomain: process.env.NHOST_SUBDOMAIN,
});

export default async (req: Request, res: Response) => {
  const authorization = req.headers.authorization ?? "";
  if (!authorization) {
    return res.status(401).json({ error: "Missing Authorization header" });
  }

  const { body } = await nhost.graphql.request<{ notes: { title: string; content: string }[] }>(
    { query: `query { notes(order_by: {updated_at: desc}) { title content } }` },
    { headers: { Authorization: authorization } },
  );

  const notes = body.data?.notes ?? [];
  const document =
    `# Notes export\n\n_${notes.length} note(s)_\n\n` +
    notes.map((n) => `# ${n.title || "Untitled"}\n\n${n.content}`).join("\n\n---\n\n");

  const fileName = `export-${Date.now()}.md`;
  const upload = await nhost.storage.uploadFiles(
    { "bucket-id": "notes", "file[]": [new File([document], fileName, { type: "text/markdown" })] },
    { headers: { Authorization: authorization } },
  );

  const file = upload.body.processedFiles?.[0];
  res.status(200).json({ fileId: file?.id, fileName, noteCount: notes.length });
};
```

The function imports `@nhost/nhost-js`, and the functions runtime bundles each
function with esbuild against the `functions/` project's own dependencies — so
install the SDK there (the `functions/` directory `nhost init` scaffolded in
Part 1) or the build fails with `Could not resolve "@nhost/nhost-js"`. Pin the
v5 prerelease that `export.ts` targets — npm's `latest` is still the v4 line,
which lacks the `createClient` API the function uses:

```bash Terminal
cd functions
npm install @nhost/nhost-js@5.0.0-beta11
```

## Call it from the CLI

`Functions.Post` sends an authenticated `POST` — the SDK attaches your token, so
the function receives it in the `Authorization` header:

```go title="main.go"
func cmdExport(ctx context.Context, c *nhost.Client) error {
	resp, _, err := c.Functions.Post(ctx, "/notes/export", struct{}{}, nil)
	if err != nil {
		return err
	}
	out, _ := json.MarshalIndent(resp, "", "  ")
	fmt.Println(string(out))
	return nil
}
```

Register it on the root command, plus add the `encoding/json` import:

// newClient builds the Nhost client. The client middleware attaches the access
// token to every request and refreshes it automatically once we sign in.
func newClient() *nhost.Client {
	return nhost.New(nhost.Options{
		Subdomain: env("NHOST_SUBDOMAIN", "local"),
		Region:    env("NHOST_REGION", "local"),
		Storage:   &session.FileStorage{Path: sessionPath()},
	})
}

func sessionPath() string {
	if p := os.Getenv("NHOST_NOTES_SESSION"); p != "" {
		return p
	}
	dir, err := os.UserConfigDir()
	if err != nil {
		dir = os.TempDir()
	}
	return filepath.Join(dir, "nhost-notes", "session.json")
}

func env(key, fallback string) string {
	if v := os.Getenv(key); v != "" {
		return v
	}
	return fallback
}

func gql(ctx context.Context, c *nhost.Client, query string, vars graphql.Variables) (map[string]any, error) {
	resp, _, err := c.GraphQL.Request(ctx, query, vars, "", nil)
	if err != nil {
		return nil, err
	}
	return resp.Data, nil
}

// tiny helpers for reading the untyped response map
func dig(m map[string]any, keys ...string) any {
	var cur any = m
	for _, k := range keys {
		mm, ok := cur.(map[string]any)
		if !ok {
			return nil
		}
		cur = mm[k]
	}
	return cur
}

func str(v any) string    { s, _ := v.(string); return s }
func asSlice(v any) []any { s, _ := v.([]any); return s }

func cmdSignup(ctx context.Context, c *nhost.Client, email, password string) error {
	if _, _, err := c.Auth.SignUpEmailPassword(ctx, auth.SignUpEmailPasswordRequest{
		Email:    email,
		Password: password,
	}, nil); err != nil {
		return err
	}
	if _, ok := c.GetUserSession(); ok {
		fmt.Println("signed up and logged in as", email)
	} else {
		fmt.Println("signed up; verify your email, then `login`")
	}
	return nil
}

func cmdLogin(ctx context.Context, c *nhost.Client, email, password string) error {
	if _, _, err := c.Auth.SignInEmailPassword(ctx, auth.SignInEmailPasswordRequest{
		Email:    email,
		Password: password,
	}, nil); err != nil {
		return err
	}
	fmt.Println("logged in as", email)
	return nil
}

func cmdLogout(ctx context.Context, c *nhost.Client) error {
	if s, ok := c.GetUserSession(); ok {
		rt := s.RefreshToken
		_, _, _ = c.Auth.SignOut(ctx, auth.SignOutRequest{RefreshToken: &rt}, nil)
	}
	c.ClearSession()
	fmt.Println("logged out")
	return nil
}

func cmdWhoami(c *nhost.Client) error {
	s, ok := c.GetUserSession()
	if !ok || s.User == nil {
		return errors.New("not logged in")
	}
	email := ""
	if s.User.Email != nil {
		email = *s.User.Email
	}
	fmt.Printf("%s (%s)\n", email, s.User.ID)
	return nil
}

func noteNew(ctx context.Context, c *nhost.Client, title, content string) error {
	obj := map[string]any{"title": title, "content": content}
	data, err := gql(ctx, c, `
		mutation NewNote($obj: notes_insert_input!) {
			insert_notes_one(object: $obj) { id }
		}`, graphql.Variables{"obj": obj})
	if err != nil {
		return err
	}
	fmt.Println("created", str(dig(data, "insert_notes_one", "id")))
	return nil
}

func noteLs(ctx context.Context, c *nhost.Client) error {
	data, err := gql(ctx, c, `
		query {
			notes(order_by: [{is_pinned: desc}, {updated_at: desc}]) {
				id title is_pinned
			}
		}`, nil)
	if err != nil {
		return err
	}
	notes := asSlice(data["notes"])
	if len(notes) == 0 {
		fmt.Println("(no notes)")
		return nil
	}
	for _, n := range notes {
		m, _ := n.(map[string]any)
		pin := " "
		if b, _ := m["is_pinned"].(bool); b {
			pin = "*"
		}
		fmt.Printf("%s %s  %s\n", pin, str(m["id"]), str(m["title"]))
	}
	return nil
}

func updateNote(ctx context.Context, c *nhost.Client, id string, set map[string]any) error {
	data, err := gql(ctx, c, `
		mutation UpdateNote($id: uuid!, $set: notes_set_input!) {
			update_notes_by_pk(pk_columns: {id: $id}, _set: $set) { id }
		}`, graphql.Variables{"id": id, "set": set})
	if err != nil {
		return err
	}
	if dig(data, "update_notes_by_pk", "id") == nil {
		return errors.New("note not found or not permitted")
	}
	fmt.Println("updated", id)
	return nil
}

func upsertTag(ctx context.Context, c *nhost.Client, name string) (string, error) {
	data, err := gql(ctx, c, `
		mutation UpsertTag($name: String!) {
			insert_tags_one(
				object: {name: $name}
				on_conflict: {constraint: tags_user_id_name_key, update_columns: [name]}
			) { id }
		}`, graphql.Variables{"name": name})
	if err != nil {
		return "", err
	}
	id := str(dig(data, "insert_tags_one", "id"))
	if id == "" {
		return "", errors.New("could not create tag")
	}
	return id, nil
}

func noteTag(ctx context.Context, c *nhost.Client, noteID, tagName string) error {
	tagID, err := upsertTag(ctx, c, tagName)
	if err != nil {
		return err
	}
	_, err = gql(ctx, c, `
		mutation TagNote($noteId: uuid!, $tagId: uuid!) {
			insert_note_tags_one(
				object: {note_id: $noteId, tag_id: $tagId}
				on_conflict: {constraint: note_tags_pkey, update_columns: []}
			) { note_id }
		}`, graphql.Variables{"noteId": noteID, "tagId": tagID})
	if err != nil {
		return err
	}
	fmt.Printf("tagged %s with #%s\n", noteID, tagName)
	return nil
}

func cmdAttach(ctx context.Context, c *nhost.Client, noteID, file string) error {
	raw, err := os.ReadFile(file)
	if err != nil {
		return err
	}
	name := filepath.Base(file)
	b := bucket
	up, _, err := c.Storage.UploadFiles(ctx, storage.UploadFilesBody{
		BucketID: &b,
		File:     [][]byte{raw},
		Metadata: &[]storage.UploadFileMetadata{{Name: &name}},
	}, nil)
	if err != nil {
		return err
	}
	if len(up.ProcessedFiles) == 0 {
		return errors.New("upload failed")
	}
	fileID := up.ProcessedFiles[0].ID

	if _, err := gql(ctx, c, `
		mutation Attach($noteId: uuid!, $fileId: uuid!) {
			insert_note_attachments_one(object: {note_id: $noteId, file_id: $fileId}) { file_id }
		}`, graphql.Variables{"noteId": noteID, "fileId": fileID}); err != nil {
		return err
	}
	fmt.Printf("attached %s (file %s) to %s\n", name, fileID, noteID)
	return nil
}

func cmdDownload(ctx context.Context, c *nhost.Client, fileID, outPath string) error {
	resp, _, err := c.Storage.GetFile(ctx, fileID, nil, nil)
	if err != nil {
		return err
	}
	if err := os.WriteFile(outPath, resp, 0o600); err != nil {
		return err
	}
	fmt.Printf("wrote %d bytes to %s\n", len(resp), outPath)
	return nil
}

func cmdExport(ctx context.Context, c *nhost.Client) error {
	resp, _, err := c.Functions.Post(ctx, "/notes/export", struct{}{}, nil)
	if err != nil {
		return err
	}
	out, _ := json.MarshalIndent(resp, "", "  ")
	fmt.Println(string(out))
	return nil
}

# You need a second user to share with. Create one in its own session file so it

## What you built

You now have a working notes CLI that exercises every core Nhost product:

- **Auth** with a persisted session that survives between runs
- **GraphQL** CRUD protected by owner-scoped row permissions
- **Storage** for attachments, access-controlled by the same rules
- **Functions** that reuse the caller's identity to combine products server-side

You can extend it with `show`, `edit`/`mv`/`rm`, a
`notebook new`/`notebook ls` group, a `tag ls`/`new`/`add`/`rm` group, and
richer output using the same patterns you've applied here.
